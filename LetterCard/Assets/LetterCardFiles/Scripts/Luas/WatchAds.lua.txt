local M = {}

function M.BindRelifeButton(button, canvasInstance)
    button.onClick:AddListener(function()
        print("按钮点击了，调用 ContinueChallenge")
        canvasInstance:ContinueChallenge()
    end)
end

function M.BindGetTipButton(button, getReward)
    button.onClick:AddListener(function()
        print("按钮点击了，调用 ContinueChallenge")
        getReward:ShareForWordTip()
    end)
end
